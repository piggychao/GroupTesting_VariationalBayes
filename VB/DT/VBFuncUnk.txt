# Compute the log-determinant of a matrix
ldet <- function(X) {
  if(!is.matrix(X)) return(log(X))
   determinant(X,logarithm = TRUE)$modulus
  }

########## VB - function: Beta ###############
VB_beta <- function(input_x, input_y, input_omega, input_prior){
  PSigma_inv <- input_prior$Sigma_inv 
  Pmu_Sigma <- input_prior$mu_Sigma

  var_beta_inv <- crossprod(input_x*input_omega, input_x) + PSigma_inv
  var_beta <- solve(var_beta_inv)
  m_ome <- crossprod(input_x, input_y-1/2)
  mu_beta <- var_beta %*% (m_ome + Pmu_Sigma)
  return(cbind(mu_beta,var_beta))
}

### when data_z=0
cal_prob0 <- function(a, b, c, x_abc, input_beta, 
                      Se1_alpha, Se1_beta, Sp1_alpha, Sp1_beta){
  pool_abc <- a + b + c
  log_1minSe1 <- digamma(Se1_beta) - digamma(Se1_alpha+Se1_beta)
  log_Sp1 <- digamma(Sp1_alpha) - digamma(Sp1_alpha+Sp1_beta)
  
  p1 <- exp((pool_abc>0)*log_1minSe1 + (1-(pool_abc>0))*log_Sp1 + 
              a*(x_abc[1,] %*% input_beta) + 
              b*(x_abc[2,] %*% input_beta) +
              c*(x_abc[3,] %*% input_beta))
  return(p1)
}

VB_ytilde0 <- function(input_x_sub, input_mu_beta,
                       input_Se1_alpha, input_Se1_beta, 
                       input_Sp1_alpha, input_Sp1_beta){
  x_sub <- input_x_sub
  ## Calculate p_abc for all 8 possible combinations under a pool size of 3
  p_sub1 <- cal_prob0(0, 0, 0, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  p_sub2 <- cal_prob0(1, 0, 0, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  p_sub3 <- cal_prob0(0, 1, 0, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  p_sub4 <- cal_prob0(1, 1, 0, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  p_sub5 <- cal_prob0(0, 0, 1, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  p_sub6 <- cal_prob0(1, 0, 1, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  p_sub7 <- cal_prob0(0, 1, 1, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  p_sub8 <- cal_prob0(1, 1, 1, x_sub, input_mu_beta, input_Se1_alpha, input_Se1_beta, input_Sp1_alpha, input_Sp1_beta)
  
  p_denominator <- sum(p_sub1, p_sub2, p_sub3, p_sub4,p_sub5, p_sub6, p_sub7, p_sub8)
  
  p_cal1 <- sum(p_sub2, p_sub4, p_sub6, p_sub8)/p_denominator
  p_cal2 <- sum(p_sub3, p_sub4, p_sub7, p_sub8)/p_denominator
  p_cal3 <- sum(p_sub5, p_sub6, p_sub7, p_sub8)/p_denominator
  p_cal0 <- p_sub1/p_denominator
  
  return(c(p_cal1, p_cal2, p_cal3, p_cal0))
}

### when data_z=1 
cal_prob1 <- function(a, b, c, y_abc, x_abc, input_beta, 
                      Se1_alpha, Se1_beta, Se2_alpha, Se2_beta,
                      Sp1_alpha, Sp1_beta, Sp2_alpha, Sp2_beta){
  pool_abc <- a + b + c
  log_Se1 <- digamma(Se1_alpha) - digamma(Se1_alpha + Se1_beta)
  log_1minSp1 <- digamma(Sp1_beta) - digamma(Sp1_alpha + Sp1_beta)
  log_Se2 <- digamma(Se2_alpha) - digamma(Se2_alpha + Se2_beta)
  log_1minSe2 <- digamma(Se2_beta) - digamma(Se2_alpha + Se2_beta)
  log_Sp2 <- digamma(Sp2_alpha) - digamma(Sp2_alpha + Sp2_beta)
  log_1minSp2 <- digamma(Sp2_beta) - digamma(Sp2_alpha + Sp2_beta)
  
  p1 <- exp((pool_abc>0)*log_Se1 + (1-(pool_abc>0))*log_1minSp1 + 
              (a==1)*(y_abc[1]*log_Se2 + (1-y_abc[1])*log_1minSe2) + 
              (1-(a==1))*((1-y_abc[1])*log_Sp2 + y_abc[1]*log_1minSp2) + 
              (b==1)*(y_abc[2]*log_Se2 + (1-y_abc[2])*log_1minSe2) + 
              (1-(b==1))*((1-y_abc[2])*log_Sp2 + y_abc[2]*log_1minSp2) + 
              (c==1)*(y_abc[3]*log_Se2 + (1-y_abc[3])*log_1minSe2) + 
              (1-(c==1))*((1-y_abc[3])*log_Sp2 + y_abc[3]*log_1minSp2) + 
              a*(x_abc[1,] %*% input_beta) + 
              b*(x_abc[2,] %*% input_beta) +
              c*(x_abc[3,] %*% input_beta))
  return(p1)
}

  p_denominator <- sum(p_sub1, p_sub2, p_sub3, p_sub4,p_sub5, p_sub6, p_sub7, p_sub8)
  
  p_cal1 <- sum(p_sub2, p_sub4, p_sub6, p_sub8)/p_denominator
  p_cal2 <- sum(p_sub3, p_sub4, p_sub7, p_sub8)/p_denominator
  p_cal3 <- sum(p_sub5, p_sub6, p_sub7, p_sub8)/p_denominator
  p_cal0 <- p_sub1/p_denominator
  
  return(c(p_cal1, p_cal2, p_cal3, p_cal0))
}

############ VB_Se Function ######################
VB_Se <- function(data_pool, tilde_pool, Prior_alpha, Prior_beta){
  Post_alpha_Se <- sum(data_pool * tilde_pool) + Prior_alpha
  Post_beta_Se <- sum((1-data_pool) * tilde_pool) + Prior_beta
  Se_output <- (Post_alpha_Se)/(Post_alpha_Se + Post_beta_Se) 
  return(c(Se_output, Post_alpha_Se, Post_beta_Se))
}

############ VB_Sp Function ######################
VB_Sp <- function(data_pool, tilde_pool, Prior_alpha, Prior_beta){
  Post_alpha_Sp <- sum((1-data_pool) * (1-tilde_pool)) + Prior_alpha
  Post_beta_Sp <- sum(data_pool * (1-tilde_pool)) + Prior_beta
  Sp_output <- (Post_alpha_Sp)/(Post_alpha_Sp + Post_beta_Sp) 
  return(c(Sp_output, Post_alpha_Sp, Post_beta_Sp))
}

############# Calculate Beta Density Function #####
cal_prob_beta <- function(alpha_value, beta_value){
  gamma_cons <- lbeta(alpha_value, beta_value)
  log_beta <- digamma(alpha_value) - digamma(alpha_value + beta_value)
  log_1minbeta <- digamma(beta_value) - digamma(alpha_value + beta_value)
  kernel <- (alpha_value-1)*log_beta + (beta_value-1)*log_1minbeta
  return(-gamma_cons + kernel)
}

############ Calculate Y_tilde Joint Density Function #####
cal_density_Y0 <- function(Ey1, Ey2, Ey3, x_abc, input_beta, 
                           Se1_alpha, Se1_beta, Sp1_alpha, Sp1_beta){
  pool_abc <- 1-(1-Ey1)*(1-Ey2)*(1-Ey3)
  log_1minSe1 <- digamma(Se1_beta) - digamma(Se1_alpha+Se1_beta)
  log_Sp1 <- digamma(Sp1_alpha) - digamma(Sp1_alpha+Sp1_beta)
  
  p1 <- (pool_abc)*log_1minSe1 + (1-pool_abc)*log_Sp1 + 
    Ey1*(x_abc[1,] %*% input_beta) + 
    Ey2*(x_abc[2,] %*% input_beta) +
    Ey3*(x_abc[3,] %*% input_beta)
  return(p1)
}

cal_density_Y1 <- function(Ey1, Ey2, Ey3, y_abc, x_abc, input_beta, 
                           Se1_alpha, Se1_beta, Se2_alpha, Se2_beta,
                           Sp1_alpha, Sp1_beta, Sp2_alpha, Sp2_beta){
  pool_abc <- 1-(1-Ey1)*(1-Ey2)*(1-Ey3)
  log_Se1 <- digamma(Se1_alpha) - digamma(Se1_alpha + Se1_beta)
  log_1minSp1 <- digamma(Sp1_beta) - digamma(Sp1_alpha + Sp1_beta)
  log_Se2 <- digamma(Se2_alpha) - digamma(Se2_alpha + Se2_beta)
  log_1minSe2 <- digamma(Se2_beta) - digamma(Se2_alpha + Se2_beta)
  log_Sp2 <- digamma(Sp2_alpha) - digamma(Sp2_alpha + Sp2_beta)
  log_1minSp2 <- digamma(Sp2_beta) - digamma(Sp2_alpha + Sp2_beta)
  
  p1 <- pool_abc*log_Se1 + (1-pool_abc)*log_1minSp1 + 
    Ey1*(y_abc[1]*log_Se2 + (1-y_abc[1])*log_1minSe2) + 
    (1-Ey1)*((1-y_abc[1])*log_Sp2 + y_abc[1]*log_1minSp2) + 
    Ey2*(y_abc[2]*log_Se2 + (1-y_abc[2])*log_1minSe2) + 
    (1-Ey2)*((1-y_abc[2])*log_Sp2 + y_abc[2]*log_1minSp2) + 
    Ey3*(y_abc[3]*log_Se2 + (1-y_abc[3])*log_1minSe2) + 
    (1-Ey3)*((1-y_abc[3])*log_Sp2 + y_abc[3]*log_1minSp2) + 
    Ey1*(x_abc[1,] %*% input_beta) + 
    Ey2*(x_abc[2,] %*% input_beta) +
    Ey3*(x_abc[3,] %*% input_beta)
  return(p1)
}

product_abc <- function(a, b, c) {
    return(a * b * c)
  }
  
#function: 1/(x*(1-x))
x_x1mins <- function(x){
    return(1/(x * (1-x)))
  }

## H Manually Calculation 
## - expectation of second derivatives under unknown Se&Sp
cal_Hessian_unknw <- function(mean_beta, mean_y, 
                              input_x, input_z, input_retest_y,
                              input_Se, input_Sp,
                              retest_ind){
  p_est <- plogis(input_x %*% mean_beta)
  pq_est <- p_est*(1-p_est)
  
  J <- length(input_z)
  Ez.mtx <- matrix(1-mean_y, nrow = J, byrow=T) 
  Ez_1mins <- mapply(product_abc, Ez.mtx[,1], Ez.mtx[,2], Ez.mtx[,3]) # E(Z_tilde=0)=1-E(Z_tilde=1)
  Ez <- 1-Ez_1mins # E(Z_tilde=1)
  
  Ey_sub <- mean_y[retest_ind] # E(y_tilde) from 1 to J* (in positive pools)
  Ey_sub_1mins <- 1-Ey_sub
  
  retest_y_sub <- input_retest_y[retest_ind]
  Ez_sub <- Ez[which(input_z>0)] # E(z_tilde) for positive pools only
  Ez_sub_1mins <- 1-Ez_sub
  
  p1 <- sum(-pq_est)
  p2 <- sum(-input_x[,2]^2*pq_est)
  p3 <- sum(-input_x[,3]^2*pq_est)
  
  p4 <- x_x1mins(input_Se[1])^2 * sum(-Ez*(input_Se[1]^2+input_z-2*input_z*input_Se[1])) #d2_Se_p
  p5 <- x_x1mins(input_Se[2])^2 * sum(-Ey_sub*(input_Se[2]^2+retest_y_sub-2*retest_y_sub*input_Se[2])) #d2_Se_i
  p6 <- x_x1mins(input_Sp[1])^2 * sum(-Ez_1mins*(input_Sp[1]^2+(1-input_z)-2*(1-input_z)*input_Sp[1])) #d2_Sp_p
  p7 <- x_x1mins(input_Sp[2])^2 * sum(-Ey_sub_1mins*(input_Sp[2]^2+(1-retest_y_sub)-2*(1-retest_y_sub)*input_Sp[2])) #d2_Sp_i
  
  p12 <- sum(-input_x[,2]*pq_est)
  p13 <- sum(-input_x[,3]*pq_est)
  p23 <- sum(-(input_x[,2]*input_x[,3])*pq_est)
  
  Hessian_manu <- matrix(NA, nrow=dim(input_x)+length(input_Se)+length(input_Sp), 
                              ncol=dim(input_x)+length(input_Se)+length(input_Sp))
  Hessian_manu <- diag(c(p1,p2,p3,p4,p5,p6,p7))
  Hessian_manu[1,2] <- Hessian_manu[2,1] <- p12
  Hessian_manu[1,3] <- Hessian_manu[3,1] <- p13
  Hessian_manu[2,3] <- Hessian_manu[3,2] <- p23
  
  return(Hessian_manu)
}

## Summations of Covariance_y1y2 * multipliers (1 to N)
cal_cov_sums_unknw <- function(mean_beta, mean_y, 
                               input_x, input_z, input_retest_y,
                               input_Se, input_Sp,
                               input_Se_alp_beta, input_Sp_alp_beta,
                               input_pool_index){
  J <- length(input_z)
  size <- dim(input_pool_index)[2]
  p <- dim(input_x)[2]
  combinations <- expand.grid(c(0, 1), c(0, 1), c(0, 1))
  
  Ez.mtx <- matrix(1-mean_y, nrow = J, byrow=T) 
  Ez_1mins <- mapply(product_abc, Ez.mtx[,1], Ez.mtx[,2], Ez.mtx[,3]) # E(Z_tilde=0)=1-E(Z_tilde=1)
  Ez <- 1-Ez_1mins # E(Z_tilde=1)
  
  # Calculate cov(y_i,y_j) = E(y_i*y_j) - E(y_i)E(y_j)
  Eyy_cov <- matrix(NA, J, size)
  
  Eyy_cov_sum11 <- matrix(NA, J, size) #beta0 vs. beta0
  Eyy_cov_sum22 <- matrix(NA, J, size) #beta1 vs. beta1
  Eyy_cov_sum33 <- matrix(NA, J, size) #beta2 vs. beta2
  
  Eyy_cov_sum44 <- rep(NA, J) #Var(Se_i)
  Eyy_cov_sum55 <- rep(NA, J) #Var(Sp_i)
  
  Eyy_cov_sum1 <- matrix(NA, J, size) #beta0 vs. beta1
  Eyy_cov_sum2 <- matrix(NA, J, size) #beta0 vs. beta2
  Eyy_cov_sum3 <- matrix(NA, J, size) #beta1 vs. beta2
  
  ## Cov(Se_p, par:beta0, beta1, beta2)
  cov_Se1par_sum <- matrix(NA, nrow=J, ncol=p)
  ## Cov(Sp_p, par:beta0, beta1, beta2
  cov_Sp1par_sum <- matrix(NA, nrow=J, ncol=p)
  ## Cov(Se_i, par:beta0, beta1, beta2)
  cov_Se2par_sum1 <- rep(NA, J)
  cov_Se2par_sum2 <- rep(NA, J)
  cov_Se2par_sum3 <- rep(NA, J)
  ## Cov(Sp_i, par:beta0, beta1, beta2)
  cov_Sp2par_sum1 <- rep(NA, J)
  cov_Sp2par_sum2 <- rep(NA, J)
  cov_Sp2par_sum3 <- rep(NA, J)
  
  cov_Sp2Se2_sum <- rep(NA, J) #Cov(Se_i vs. Sp_i)
  
  for (j in 1:J){
    pool_index <- input_pool_index[j,]
    x_sub <- input_x[pool_index,]
    Ey_spool <- mean_y[pool_index]
    
    cov_Se1par_sum[j,] <- (input_z[j]-input_Se[1])*colSums(x_sub*Ey_spool*Ez_1mins[j])
    cov_Sp1par_sum[j,] <- (1-input_z[j]-input_Sp[1])*colSums(x_sub*Ey_spool*(-Ez_1mins[j]))
    
    if (input_z[j] == 0){
      p_sub <- apply(combinations, 1, 
                     function(row) cal_prob0(row[1], row[2], row[3],
                                             x_sub, mean_beta, 
                                             input_Se_alp_beta[1], input_Se_alp_beta[2],
                                             input_Sp_alp_beta[1], input_Sp_alp_beta[2]))
      p_denominator <- sum(p_sub) 
      
      Eyy.12 <- sum(p_sub[4], p_sub[8])/p_denominator #E(y1*y2)
      Eyy.13 <- sum(p_sub[6], p_sub[8])/p_denominator #E(y1*y3)
      Eyy.23 <- sum(p_sub[7], p_sub[8])/p_denominator #E(y2*y3)
      
      Eyy_cov[j,1] <- Eyy.12 - Ey_spool[1]*Ey_spool[2] #E(y1*y2)-E(y1)E(y2)
      Eyy_cov[j,2] <- Eyy.13 - Ey_spool[1]*Ey_spool[3] #E(y1*y3)-E(y1)E(y3)
      Eyy_cov[j,3] <- Eyy.23 - Ey_spool[2]*Ey_spool[3] #E(y2*y3)-E(y2)E(y3)
     }
    if (input_z[j] == 1){
      y_sub <- input_retest_y[pool_index]
      p_sub <- apply(combinations, 1, 
                     function(row) cal_prob1(row[1], row[2], row[3], 
                                             y_sub, x_sub, mean_beta, 
                                             input_Se_alp_beta[1], input_Se_alp_beta[2],
                                             input_Se_alp_beta[3], input_Se_alp_beta[4],
                                             input_Sp_alp_beta[1], input_Sp_alp_beta[2],
                                             input_Sp_alp_beta[3], input_Sp_alp_beta[4]))
      p_denominator <- sum(p_sub) 
      
      Eyy.12 <- sum(p_sub[4], p_sub[8])/p_denominator #E(y1*y2)
      Eyy.13 <- sum(p_sub[6], p_sub[8])/p_denominator #E(y1*y3)
      Eyy.23 <- sum(p_sub[7], p_sub[8])/p_denominator #E(y2*y3)
      
      Eyy_cov[j,1] <- Eyy.12 - Ey_spool[1]*Ey_spool[2] #E(y1*y2)-E(y1)E(y2)
      Eyy_cov[j,2] <- Eyy.13 - Ey_spool[1]*Ey_spool[3] #E(y1*y3)-E(y1)E(y3)
      Eyy_cov[j,3] <- Eyy.23 - Ey_spool[2]*Ey_spool[3] #E(y2*y3)-E(y2)E(y3)
      
      Eyy_cov_sum44[j] <- 2*(y_sub[1]-input_Se[2])*(y_sub[2]-input_Se[2])*Eyy_cov[j,1] + 
                          2*(y_sub[1]-input_Se[2])*(y_sub[3]-input_Se[2])*Eyy_cov[j,2] +
                          2*(y_sub[2]-input_Se[2])*(y_sub[3]-input_Se[2])*Eyy_cov[j,3]
      
      Eyy_cov_sum55[j] <- 2*(1-y_sub[1]-input_Sp[2])*(1-y_sub[2]-input_Sp[2])*Eyy_cov[j,1] + 
                          2*(1-y_sub[1]-input_Sp[2])*(1-y_sub[3]-input_Sp[2])*Eyy_cov[j,2] +
                          2*(1-y_sub[2]-input_Sp[2])*(1-y_sub[3]-input_Sp[2])*Eyy_cov[j,3]
      
      cov_Se2par_sum1[j] <- ((y_sub[1]-input_Se[2])+(y_sub[2]-input_Se[2]))*Eyy_cov[j,1] +
                            ((y_sub[1]-input_Se[2])+(y_sub[3]-input_Se[2]))*Eyy_cov[j,2] +
                            ((y_sub[2]-input_Se[2])+(y_sub[3]-input_Se[2]))*Eyy_cov[j,3] +
                            sum((y_sub-input_Se[2])*Ey_spool*(1-Ey_spool))
      cov_Se2par_sum2[j] <- ((y_sub[1]-input_Se[2])*x_sub[2,2]+(y_sub[2]-input_Se[2])*x_sub[1,2])*Eyy_cov[j,1] +
                            ((y_sub[1]-input_Se[2])*x_sub[3,2]+(y_sub[3]-input_Se[2])*x_sub[1,2])*Eyy_cov[j,2] +
                            ((y_sub[2]-input_Se[2])*x_sub[3,2]+(y_sub[3]-input_Se[2])*x_sub[2,2])*Eyy_cov[j,3] +
                            sum((y_sub-input_Se[2])*x_sub[,2]*Ey_spool*(1-Ey_spool))
      cov_Se2par_sum3[j] <- ((y_sub[1]-input_Se[2])*x_sub[2,3]+(y_sub[2]-input_Se[2])*x_sub[1,3])*Eyy_cov[j,1] +
                            ((y_sub[1]-input_Se[2])*x_sub[3,3]+(y_sub[3]-input_Se[2])*x_sub[1,3])*Eyy_cov[j,2] +
                            ((y_sub[2]-input_Se[2])*x_sub[3,3]+(y_sub[3]-input_Se[2])*x_sub[2,3])*Eyy_cov[j,3] +
                            sum((y_sub-input_Se[2])*x_sub[,3]*Ey_spool*(1-Ey_spool))
      
      cov_Sp2par_sum1[j] <- ((1-y_sub[1]-input_Sp[2])+(1-y_sub[2]-input_Sp[2]))*(-Eyy_cov[j,1]) +
                            ((1-y_sub[1]-input_Sp[2])+(1-y_sub[3]-input_Sp[2]))*(-Eyy_cov[j,2]) +
                            ((1-y_sub[2]-input_Sp[2])+(1-y_sub[3]-input_Sp[2]))*(-Eyy_cov[j,3]) +
                            sum((1-y_sub-input_Sp[2])*(-Ey_spool*(1-Ey_spool)))
      cov_Sp2par_sum2[j] <- ((1-y_sub[1]-input_Sp[2])*x_sub[2,2]+(1-y_sub[2]-input_Sp[2])*x_sub[1,2])*(-Eyy_cov[j,1]) +
                            ((1-y_sub[1]-input_Sp[2])*x_sub[3,2]+(1-y_sub[3]-input_Sp[2])*x_sub[1,2])*(-Eyy_cov[j,2]) +
                            ((1-y_sub[2]-input_Sp[2])*x_sub[3,2]+(1-y_sub[3]-input_Sp[2])*x_sub[2,2])*(-Eyy_cov[j,3]) +
                            sum((1-y_sub-input_Sp[2])*x_sub[,2]*(-Ey_spool*(1-Ey_spool)))
      cov_Sp2par_sum3[j] <- ((1-y_sub[1]-input_Sp[2])*x_sub[2,3]+(1-y_sub[2]-input_Sp[2])*x_sub[1,3])*(-Eyy_cov[j,1]) +
                            ((1-y_sub[1]-input_Sp[2])*x_sub[3,3]+(1-y_sub[3]-input_Sp[2])*x_sub[1,3])*(-Eyy_cov[j,2]) +
                            ((1-y_sub[2]-input_Sp[2])*x_sub[3,3]+(1-y_sub[3]-input_Sp[2])*x_sub[2,3])*(-Eyy_cov[j,3]) +
                            sum((1-y_sub-input_Sp[2])*x_sub[,3]*(-Ey_spool*(1-Ey_spool)))
      
      cov_Sp2Se2_sum[j] <- ((Ey_spool[1]-input_Se[2])*(1-Ey_spool[2]-input_Sp[2])+(Ey_spool[2]-input_Se[2])*(1-Ey_spool[1]-input_Sp[2]))*(-Eyy_cov[j,1]) +
                           ((Ey_spool[1]-input_Se[2])*(1-Ey_spool[3]-input_Sp[2])+(Ey_spool[3]-input_Se[2])*(1-Ey_spool[1]-input_Sp[2]))*(-Eyy_cov[j,2]) +
                           ((Ey_spool[2]-input_Se[2])*(1-Ey_spool[3]-input_Sp[2])+(Ey_spool[3]-input_Se[2])*(1-Ey_spool[2]-input_Sp[2]))*(-Eyy_cov[j,3])
      
    }
    
    x_sub_multi <- matrix(c(x_sub[1,]*x_sub[2,] + x_sub[2,]*x_sub[1,],
                            x_sub[1,]*x_sub[3,] + x_sub[3,]*x_sub[1,],
                            x_sub[2,]*x_sub[3,] + x_sub[3,]*x_sub[2,]), nrow = size)
    
    Eyy_cov_sum11[j,] <- x_sub_multi[1,] * Eyy_cov[j,]
    Eyy_cov_sum22[j,] <- x_sub_multi[2,] * Eyy_cov[j,]
    Eyy_cov_sum33[j,] <- x_sub_multi[3,] * Eyy_cov[j,]
    
    Eyy_cov_sum1[j,1] <- (x_sub[1,2]+x_sub[2,2])*Eyy_cov[j,1]
    Eyy_cov_sum1[j,2] <- (x_sub[1,2]+x_sub[3,2])*Eyy_cov[j,2]
    Eyy_cov_sum1[j,3] <- (x_sub[2,2]+x_sub[3,2])*Eyy_cov[j,3]
    
    Eyy_cov_sum2[j,1] <- (x_sub[1,3]+x_sub[2,3])*Eyy_cov[j,1]
    Eyy_cov_sum2[j,2] <- (x_sub[1,3]+x_sub[3,3])*Eyy_cov[j,2]
    Eyy_cov_sum2[j,3] <- (x_sub[2,3]+x_sub[3,3])*Eyy_cov[j,3]
    
    Eyy_cov_sum3[j,1] <- (x_sub[1,2]*x_sub[2,3]+x_sub[2,2]*x_sub[1,3])*Eyy_cov[j,1]
    Eyy_cov_sum3[j,2] <- (x_sub[1,2]*x_sub[3,3]+x_sub[3,2]*x_sub[1,3])*Eyy_cov[j,2]
    Eyy_cov_sum3[j,3] <- (x_sub[2,3]*x_sub[3,3]+x_sub[3,2]*x_sub[2,3])*Eyy_cov[j,3]
    
  }
  return(c(sum(Eyy_cov_sum11), sum(Eyy_cov_sum22), sum(Eyy_cov_sum33),
           sum(Eyy_cov_sum44, na.rm = T), sum(Eyy_cov_sum55, na.rm = T),
           sum(Eyy_cov_sum1), sum(Eyy_cov_sum2), sum(Eyy_cov_sum3),
           colSums(cov_Se1par_sum), colSums(cov_Sp1par_sum),
           sum(cov_Se2par_sum1, na.rm = T), sum(cov_Se2par_sum2, na.rm = T), sum(cov_Se2par_sum3, na.rm = T),
           sum(cov_Sp2par_sum1, na.rm = T), sum(cov_Sp2par_sum2, na.rm = T), sum(cov_Sp2par_sum3, na.rm = T),
           sum(cov_Sp2Se2_sum, na.rm = T)))
}
